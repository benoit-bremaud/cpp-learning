cmake_minimum_required(VERSION 3.16)
project(threshold_indicator LANGUAGES CXX)

add_library(threshold_indicator src/threshold_indicator.cpp)
target_include_directories(threshold_indicator PUBLIC include)
target_compile_features(threshold_indicator PUBLIC cxx_std_17)
set_target_properties(threshold_indicator PROPERTIES CXX_EXTENSIONS OFF)

include(CTest)
if(BUILD_TESTING)
    add_executable(threshold_indicator_test tests/threshold_indicator_test.cpp)
    target_link_libraries(threshold_indicator_test PRIVATE threshold_indicator)
    set_target_properties(threshold_indicator_test PROPERTIES CXX_EXTENSIONS OFF)
    add_test(NAME threshold_indicator_contract COMMAND threshold_indicator_test)
endif()

if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(threshold_indicator PRIVATE -Wall -Wextra -Wpedantic -Werror)
    if(BUILD_TESTING)
        target_compile_options(threshold_indicator_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
elseif(MSVC)
    target_compile_options(threshold_indicator PRIVATE /W4 /WX)
    if(BUILD_TESTING)
        target_compile_options(threshold_indicator_test PRIVATE /W4 /WX)
    endif()
endif()
